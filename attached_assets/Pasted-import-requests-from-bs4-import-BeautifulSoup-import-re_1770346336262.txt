import requests
from bs4 import BeautifulSoup
import re

BASE = "https://www.fnde.gov.br"
CONSULTA_URL = f"{BASE}/pddeinfo/pddeinfo/escola/consultar"

HEADERS = {
    "User-Agent": "Mozilla/5.0 (compatible; PDDEInfoDash/1.0)",
    "Accept-Language": "pt-BR,pt;q=0.9",
}

def br_to_float(s: str):
    s = (s or "").strip()
    if not s:
        return None
    s = s.replace(".", "").replace(",", ".")
    try:
        return float(s)
    except:
        return None

def get_pdde_info(ano: int, co_escola: str):
    with requests.Session() as session:
        params = {
            "ano": str(ano),
            "co_escola": str(co_escola),
            "consultar": "Consultar",
        }
        r = session.get(CONSULTA_URL, params=params, headers=HEADERS, timeout=30)
        r.raise_for_status()
        r.encoding = "iso-8859-1"  # importante
        soup = BeautifulSoup(r.text, "lxml")

    # acha o bloco da escola (quando a página retorna mais de um)
    target = None
    for item in soup.select("div.escola-item"):
        # procura td com o código exato
        if any(td.get_text(strip=True) == str(co_escola) for td in item.select("td")):
            target = item
            break
    if not target:
        return {"ok": False, "error": "Escola não encontrada no retorno (verifique ano e código INEP)."}

    data = {"ok": True, "ano": ano, "co_escola": str(co_escola)}

    # helpers
    def find_table_by_caption_contains(text):
        for t in target.select("table"):
            cap = t.find("caption")
            if cap and text.lower() in cap.get_text(" ", strip=True).lower():
                return t
        return None

    # 1) Dados da escola
    t = find_table_by_caption_contains("Dados da Escola")
    if t:
        rows = t.select("tbody tr")
        # nome / UF / município / rede
        for tr in rows:
            ths = [x.get_text(" ", strip=True) for x in tr.find_all("th")]
            tds = [x.get_text(" ", strip=True) for x in tr.find_all("td")]
            line = " ".join(ths)

            if "Nome Escola" in line and len(tds) >= 2:
                # td0 costuma ser o código, td1 o nome
                if tds[0] == str(co_escola):
                    data["nome_escola"] = tds[1]

            if "UF:" in line and "Município:" in line and "Rede de Ensino:" in line and len(tds) >= 3:
                data["uf"] = tds[0]
                data["municipio"] = tds[1]
                data["rede"] = tds[2]

        # matrículas por nível
        alunos = {}
        total = None
        for tr in rows:
            tds = [x.get_text(" ", strip=True) for x in tr.find_all("td")]
            if len(tds) >= 2:
                nivel = tds[0].strip()
                qtd = tds[-1].strip()
                if nivel.upper() == "TOTAL":
                    try:
                        total = int(qtd.replace(".", "").replace(",", ""))
                    except:
                        total = None
                elif nivel:
                    try:
                        alunos[nivel] = int(qtd.replace(".", "").replace(",", ""))
                    except:
                        pass
        data["matriculas_por_nivel"] = alunos
        data["matriculas_total"] = total

    # 2) UEx
    t = find_table_by_caption_contains("Unidade Executora Própria")
    if t:
        # primeira linha: Executora / CNPJ / UF / Município
        tds = [x.get_text(" ", strip=True) for x in t.select("tbody tr td")]
        if len(tds) >= 4:
            data["uex"] = {
                "executora": tds[0],
                "cnpj": tds[1],
                "uf": tds[2],
                "municipio": tds[3],
            }

    # 3) Dados bancários
    t = find_table_by_caption_contains("Dados Bancários")
    contas = []
    if t:
        for tr in t.select("tbody tr"):
            cols = [x.get_text(" ", strip=True) for x in tr.find_all("td")]
            if len(cols) == 6:
                contas.append({
                    "programa": cols[0],
                    "banco": cols[1],
                    "agencia": cols[2],
                    "conta": cols[3],
                    "saldo": br_to_float(cols[4]),
                    "ocorrencia": cols[5] or None,
                })
    data["contas"] = contas

    # 4) Situação UEx (mandato + prestação)
    t = find_table_by_caption_contains("Situação da Unidade Executora")
    if t:
        text = t.get_text("\n", strip=True)
        # mandato: "Início do mandato:dd/mm/aaaa Fim do mandato:dd/mm/aaaa"
        m = re.search(r"Início do mandato:\s*([0-9/]+).*Fim do mandato:\s*([0-9/]+)", text)
        data["situacao_uex"] = {
            "texto": None,
            "mandato_inicio": m.group(1) if m else None,
            "mandato_fim": m.group(2) if m else None,
        }
        # pega os textos por linha (células)
        rows = t.select("tbody tr")
        for tr in rows:
            th = tr.find("th")
            td = tr.find("td")
            if not th or not td:
                continue
            k = th.get_text(" ", strip=True).lower()
            v = td.get_text(" ", strip=True)
            if "dados cadastrais" in k:
                data["situacao_uex"]["dados_cadastrais"] = v
            if "prestação de contas" in k:
                data["situacao_uex"]["prestacao_contas"] = v

    # 5) Situação EEx
    t = find_table_by_caption_contains("Situação da Entidade Executora")
    if t:
        eex = {}
        for tr in t.select("tbody tr"):
            th = tr.find("th")
            td = tr.find("td")
            if not th or not td:
                continue
            k = th.get_text(" ", strip=True).lower()
            v = td.get_text(" ", strip=True)
            if "adesão" in k:
                eex["adesao"] = v
            if "prestação de contas" in k:
                eex["prestacao_contas"] = v
        data["situacao_eex"] = eex

    # 6) Destinações (repasses)
    # tabela tem classe "destinacao"
    destinacoes = []
    dest_table = target.select_one("div.listagem.destinacao table")
    if dest_table:
        for tr in dest_table.select("tbody tr"):
            tds = [x.get_text(" ", strip=True) for x in tr.find_all(["td"])]
            # linhas de dado geralmente tem 12 colunas
            if len(tds) == 12:
                destinacoes.append({
                    "destinacao": tds[0],
                    "devido_custeio": br_to_float(tds[1]),
                    "devido_capital": br_to_float(tds[2]),
                    "devido_total": br_to_float(tds[3]),
                    "ajuste_custeio": br_to_float(tds[4]),
                    "ajuste_capital": br_to_float(tds[5]),
                    "ajuste_total": br_to_float(tds[6]),
                    "final_devido_total": br_to_float(tds[7]),
                    "pago_custeio": br_to_float(tds[8]),
                    "pago_capital": br_to_float(tds[9]),
                    "pago_total": br_to_float(tds[10]),
                    "data_ordem_pgto": tds[11] or None,
                })
    data["destinacoes"] = destinacoes

    return data


if __name__ == "__main__":
    ano = 2025
    co_escola = "22073655"  # exemplo
    result = get_pdde_info(ano, co_escola)
    print(result)  # pronto pra virar JSON no seu backend
