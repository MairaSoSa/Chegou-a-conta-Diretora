Ela:

consulta o FNDE no backend;

recebe o HTML;

extrai dados principais;

devolve JSON.

Observação importante: como o FNDE entrega HTML e a estrutura pode variar, deixei o parser resiliente, mas alguns seletores podem precisar de ajuste fino quando você testar com páginas reais.

import express from "express";
import axios from "axios";
import * as cheerio from "cheerio";

const app = express();
const PORT = process.env.PORT || 3000;

const FNDE_BASE = "https://www.fnde.gov.br";

function createHttpClient() {
  return axios.create({
    baseURL: FNDE_BASE,
    timeout: 30000,
    headers: {
      "User-Agent": "Mozilla/5.0",
      "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
      "Accept-Language": "pt-BR,pt;q=0.9,en;q=0.8"
    }
  });
}

function normalizeText(value = "") {
  return value.replace(/\s+/g, " ").replace(/\u00a0/g, " ").trim();
}

function onlyDigits(value = "") {
  return String(value).replace(/\D/g, "");
}

function parseMoney(value = "") {
  const text = normalizeText(value);
  if (!text) return null;
  const normalized = text.replace(/\./g, "").replace(",", ".");
  const num = Number(normalized);
  return Number.isNaN(num) ? null : num;
}

function parseKeyValuePairsFromText($, rootSelector = "body") {
  const text = normalizeText($(rootSelector).text());
  return text;
}

function extractSchoolBasicData(text) {
  const result = {};

  const codigo = text.match(/Cod\.\s*Escola:\s*([0-9]+)/i);
  const nome = text.match(/Nome\s*Escola:\s*(.+?)\s+UF:/i);
  const uf = text.match(/UF:\s*([A-Z]{2})/i);
  const municipio = text.match(/Munic[ií]pio:\s*(.+?)\s+Rede\s*de\s*Ensino:/i);
  const rede = text.match(/Rede\s*de\s*Ensino:\s*(.+?)(?:\s+N[ií]vel\s*Ensino|\s+Unidade\s*Executora)/i);

  result.codigo_escola = codigo?.[1] || null;
  result.nome_escola = nome ? normalizeText(nome[1]) : null;
  result.uf = uf?.[1] || null;
  result.municipio = municipio ? normalizeText(municipio[1]) : null;
  result.rede_ensino = rede ? normalizeText(rede[1]) : null;

  return result;
}

function extractMatriculas(text) {
  const rows = [];
  const blockMatch = text.match(/N[ií]vel\s*Ensino\s*Qtd\s*Alunos\s*(.+?)\s*Unidade\s*Executora\s*Pr[oó]pria/i);
  if (!blockMatch) return rows;

  const block = normalizeText(blockMatch[1]);
  const regex = /(ENSINO FUNDAMENTAL|PR[ÉE]-ESCOLAR|CRECHE|ENSINO M[EÉ]DIO|EJA|TOTAL)\s+([0-9]+)/gi;

  let match;
  while ((match = regex.exec(block)) !== null) {
    rows.push({
      nivel: normalizeText(match[1]),
      qtd: Number(match[2])
    });
  }

  return rows;
}

function extractUexData(text) {
  const result = {};

  const executora = text.match(/Executora:\s*(.+?)\s+CNPJ:/i);
  const cnpj = text.match(/CNPJ:\s*([0-9.\-\/]+)/i);
  const uf = text.match(/Executora:.*?CNPJ:.*?UF:\s*([A-Z]{2})/i);
  const municipio = text.match(/Executora:.*?Munic[ií]pio:\s*(.+?)\s+Dados\s+Banc[aá]rios/i);

  result.executora = executora ? normalizeText(executora[1]) : null;
  result.cnpj = cnpj ? normalizeText(cnpj[1]) : null;
  result.cnpj_numerico = cnpj ? onlyDigits(cnpj[1]) : null;
  result.uf = uf?.[1] || null;
  result.municipio = municipio ? normalizeText(municipio[1]) : null;

  return result;
}

function extractBankDataFromTables($) {
  const results = [];

  $("table").each((_, table) => {
    const headers = $(table)
      .find("tr")
      .first()
      .find("th, td")
      .map((__, el) => normalizeText($(el).text()))
      .get();

    const looksLikeBankTable =
      headers.some(h => /Programa\/A[cç][aã]o/i.test(h)) &&
      headers.some(h => /Banco/i.test(h)) &&
      headers.some(h => /Ag[eê]ncia/i.test(h)) &&
      headers.some(h => /Conta/i.test(h)) &&
      headers.some(h => /Saldo/i.test(h));

    if (!looksLikeBankTable) return;

    $(table)
      .find("tr")
      .slice(1)
      .each((__, row) => {
        const cols = $(row)
          .find("td")
          .map((___, el) => normalizeText($(el).text()))
          .get();

        if (cols.length >= 5) {
          results.push({
            programa: cols[0] || null,
            banco: cols[1] || null,
            agencia: cols[2] || null,
            conta: cols[3] || null,
            saldo_texto: cols[4] || null,
            saldo: parseMoney(cols[4]),
            ocorrencia: cols[5] || null
          });
        }
      });
  });

  return results;
}

function extractRepassesFromTables($) {
  const repasses = [];

  $("table").each((_, table) => {
    const headers = $(table)
      .find("tr")
      .first()
      .find("th, td")
      .map((__, el) => normalizeText($(el).text()))
      .get();

    const looksLikeRepasseTable =
      headers.some(h => /Destina[cç][aã]o/i.test(h)) &&
      headers.some(h => /Vl\s*Devido/i.test(h)) &&
      headers.some(h => /Valor\s*Pago\s*Total/i.test(h));

    if (!looksLikeRepasseTable) return;

    $(table)
      .find("tr")
      .slice(1)
      .each((__, row) => {
        const cols = $(row)
          .find("td")
          .map((___, el) => normalizeText($(el).text()))
          .get();

        if (cols.length >= 12) {
          repasses.push({
            destinacao: cols[0] || null,
            vl_devido_custeio_texto: cols[1] || null,
            vl_devido_capital_texto: cols[2] || null,
            vl_devido_total_texto: cols[3] || null,
            vl_ajuste_custeio_texto: cols[4] || null,
            vl_ajuste_capital_texto: cols[5] || null,
            vl_ajuste_total_texto: cols[6] || null,
            vl_final_devido_total_texto: cols[7] || null,
            vl_pago_custeio_texto: cols[8] || null,
            vl_pago_capital_texto: cols[9] || null,
            valor_pago_total_texto: cols[10] || null,
            data_ord_pgto: cols[11] || null,

            vl_devido_custeio: parseMoney(cols[1]),
            vl_devido_capital: parseMoney(cols[2]),
            vl_devido_total: parseMoney(cols[3]),
            vl_ajuste_custeio: parseMoney(cols[4]),
            vl_ajuste_capital: parseMoney(cols[5]),
            vl_ajuste_total: parseMoney(cols[6]),
            vl_final_devido_total: parseMoney(cols[7]),
            vl_pago_custeio: parseMoney(cols[8]),
            vl_pago_capital: parseMoney(cols[9]),
            valor_pago_total: parseMoney(cols[10])
          });
        }
      });
  });

  return repasses;
}

function extractPendencias(text, label) {
  const regex = new RegExp(`${label}[\\s\\S]*?Ano\\s*Exerc[ií]cio\\s*Programa\\s*([\\s\\S]*?)(?:Situa[cç][aã]o da Entidade Executora|Destina[cç][aã]o|$)`, "i");
  const match = text.match(regex);
  if (!match) return [];

  const block = normalizeText(match[1]);
  const results = [];
  const rowRegex = /(20[0-9]{2})\s+([A-Z0-9\-ÁÀÂÃÉÊÍÓÔÕÚÇ ]+)/gi;

  let row;
  while ((row = rowRegex.exec(block)) !== null) {
    results.push({
      ano_exercicio: Number(row[1]),
      programa: normalizeText(row[2])
    });
  }

  return results;
}

function extractSituacaoUex(text) {
  const atualizacao = text.match(/Data\s*de\s*atualiza[cç][aã]o:\s*([0-9]{2}\/[0-9]{2}\/[0-9]{4})/i);
  const mandato = text.match(/In[ií]cio\s*do\s*mandato:\s*([0-9]{2}\/[0-9]{2}\/[0-9]{4})\s*Fim\s*do\s*mandato:\s*([0-9]{2}\/[0-9]{2}\/[0-9]{4})/i);

  return {
    data_atualizacao: atualizacao?.[1] || null,
    mandato_inicio: mandato?.[1] || null,
    mandato_fim: mandato?.[2] || null,
    programas_pendentes: extractPendencias(text, "Prestação de Contas")
  };
}

function parseSchoolPage(html) {
  const $ = cheerio.load(html);
  const text = parseKeyValuePairsFromText($);

  const basico = extractSchoolBasicData(text);
  const matriculas = extractMatriculas(text);
  const uex = extractUexData(text);
  const dados_bancarios = extractBankDataFromTables($);
  const repasses = extractRepassesFromTables($);
  const situacao_uex = extractSituacaoUex(text);

  const qtd_alunos_total =
    matriculas.find(item => /TOTAL/i.test(item.nivel))?.qtd ||
    matriculas.reduce((acc, item) => (/TOTAL/i.test(item.nivel) ? acc : acc + (item.qtd || 0)), 0);

  return {
    ...basico,
    qtd_alunos_total,
    matriculas,
    uex,
    dados_bancarios,
    situacao_uex,
    repasses
  };
}

function parseSchoolListPage(html) {
  const $ = cheerio.load(html);
  const items = [];

  $("table").each((_, table) => {
    const headers = $(table)
      .find("tr")
      .first()
      .find("th, td")
      .map((__, el) => normalizeText($(el).text()))
      .get();

    const looksLikeSchoolTable =
      headers.some(h => /Escola|Nome/i.test(h)) ||
      headers.some(h => /Cod\./i.test(h));

    if (!looksLikeSchoolTable) return;

    $(table)
      .find("tr")
      .slice(1)
      .each((__, row) => {
        const cols = $(row)
          .find("td")
          .map((___, el) => normalizeText($(el).text()))
          .get();

        if (!cols.length) return;

        const joined = cols.join(" ");
        const codeMatch = joined.match(/\b([0-9]{8})\b/);

        items.push({
          codigo_escola: codeMatch?.[1] || null,
          nome_escola: cols.find(c => /[A-ZÁÀÂÃÉÊÍÓÔÕÚÇ]/i.test(c) && !/^\d+$/.test(c)) || null,
          raw: cols
        });
      });
  });

  return items.filter((item, index, arr) => {
    const key = `${item.codigo_escola || ""}-${item.nome_escola || ""}`;
    return arr.findIndex(other => `${other.codigo_escola || ""}-${other.nome_escola || ""}` === key) === index;
  });
}

function parseSaldoPage(html) {
  const $ = cheerio.load(html);
  const saldos = [];

  $("table").each((_, table) => {
    const headers = $(table)
      .find("tr")
      .first()
      .find("th, td")
      .map((__, el) => normalizeText($(el).text()))
      .get();

    const looksLikeSaldoTable =
      headers.some(h => /Programa/i.test(h)) &&
      headers.some(h => /Saldo/i.test(h));

    if (!looksLikeSaldoTable) return;

    $(table)
      .find("tr")
      .slice(1)
      .each((__, row) => {
        const cols = $(row)
          .find("td")
          .map((___, el) => normalizeText($(el).text()))
          .get();

        if (cols.length >= 2) {
          saldos.push({
            programa: cols[0] || null,
            banco: cols[1] || null,
            agencia: cols[2] || null,
            conta: cols[3] || null,
            saldo_texto: cols[4] || null,
            saldo: parseMoney(cols[4]),
            raw: cols
          });
        }
      });
  });

  return saldos;
}

app.get("/api/pdde/escola", async (req, res) => {
  const { ano, co_escola, cnpj = "", co_esfera_adm = "", sg_uf = "", co_municipio_fnde = "" } = req.query;

  if (!ano || !co_escola) {
    return res.status(400).json({
      ok: false,
      error: "Parâmetros obrigatórios: ano e co_escola."
    });
  }

  try {
    const http = createHttpClient();

    const response = await http.get("/pddeinfo/pddeinfo/escola/consultar", {
      params: {
        ano,
        co_escola,
        cnpj,
        co_esfera_adm,
        sg_uf,
        co_municipio_fnde,
        consultar: "Consultar"
      }
    });

    const parsed = parseSchoolPage(response.data);

    res.json({
      ok: true,
      fonte: "FNDE PDDE Info",
      consulta: {
        ano,
        co_escola,
        cnpj,
        co_esfera_adm,
        sg_uf,
        co_municipio_fnde
      },
      data: parsed
    });
  } catch (error) {
    res.status(500).json({
      ok: false,
      error: "Falha ao consultar a escola no FNDE.",
      detail: error.message
    });
  }
});

app.get("/api/pdde/escolas", async (req, res) => {
  const { ano, sg_uf = "", co_municipio_fnde = "", co_esfera_adm = "", cnpj = "" } = req.query;

  if (!ano) {
    return res.status(400).json({
      ok: false,
      error: "Parâmetro obrigatório: ano."
    });
  }

  try {
    const http = createHttpClient();

    const response = await http.get("/pddeinfo/pddeinfo/escola/consultar", {
      params: {
        ano,
        co_escola: "",
        cnpj,
        co_esfera_adm,
        sg_uf,
        co_municipio_fnde,
        consultar: "Consultar"
      }
    });

    const items = parseSchoolListPage(response.data);

    res.json({
      ok: true,
      fonte: "FNDE PDDE Info",
      consulta: {
        ano,
        sg_uf,
        co_municipio_fnde,
        co_esfera_adm,
        cnpj
      },
      total: items.length,
      items
    });
  } catch (error) {
    res.status(500).json({
      ok: false,
      error: "Falha ao listar escolas no FNDE.",
      detail: error.message
    });
  }
});

  const { ano, cnpj } = req.query;

  if (!ano || !cnpj) {
    return res.status(400).json({
      ok: false,
      error: "Parâmetros obrigatórios: ano e cnpj."
    });

  try {
    const http = createHttpClient();

    const response = await http.get("/pddeinfo/consultasaldoentidade/consultasaldoentidade/consultasaldoentidade", {
      params: {
        ano,
        cnpj,
        consultar: "Consultar"
      }
    });

    const saldos = parseSaldoPage(response.data);

    res.json({
      ok: true,
      fonte: "FNDE PDDE Info",

        ano,
        cnpj: onlyDigits(cnpj)

      total: saldos.length,

    res.status(500).json({
      ok: false,
      error: "Falha ao consultar saldo da UEx no FNDE.",

Exemplos de uso

Buscar escola:

curl "http://localhost:3000/api/pdde/escola?ano=2024&co_escola=15040836"

Listar escolas por município:

curl "http://localhost:3000/api/pdde/escolas?ano=2024&sg_uf=PA&co_municipio_fnde=150140&co_esfera_adm=2"

Buscar saldo da UEx:

curl "http://localhost:3000/api/pdde/uex/saldo?ano=2024&cnpj=01904042000110"
O que eu recomendo ajustar primeiro no seu teste real

Quando você subir isso, confira:

se a tabela de listagem de escolas vem mesmo na rota /api/pdde/escolas;

se o bloco de dados bancários está em tabela;

se os repasses estão em uma única tabela ou em mais de uma;

se a página retorna tudo já no HTML ou exige navegação adicional.
