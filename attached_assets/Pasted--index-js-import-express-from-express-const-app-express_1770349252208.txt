// index.js
import express from "express";

const app = express();
const PORT = process.env.PORT || 3000;

const BASE = "https://www.fnde.gov.br/sigpcadm";

/**
 * Fetch helper com headers "browser-like" para reduzir bloqueios.
 * Retorna JSON quando possível; senão retorna texto bruto.
 */
async function fndeGet(path, params = {}, extraHeaders = {}) {
  const url = new URL(BASE + path);
  for (const [k, v] of Object.entries(params)) {
    if (v !== undefined && v !== null && String(v).trim() !== "") {
      url.searchParams.set(k, String(v));
    }
  }

  const res = await fetch(url.toString(), {
    method: "GET",
    headers: {
      accept: "application/json,text/javascript,*/*;q=0.1",
      "user-agent":
        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120 Safari/537.36",
      referer: BASE + "/sistema.pu?operation=localizar",
      ...extraHeaders,
    },
  });

  const text = await res.text();
  if (!res.ok) {
    throw new Error(
      `FNDE HTTP ${res.status} ${res.statusText} | ${url.pathname}${url.search} | body: ${text.slice(
        0,
        300
      )}`
    );
  }

  try {
    return JSON.parse(text);
  } catch {
    return { raw: text };
  }
}

/** Normaliza CNPJ: só dígitos */
function normalizeCnpj(cnpj) {
  if (!cnpj) return "";
  return String(cnpj).replace(/\D/g, "");
}

/**
 * 1) LOCALIZAR PCs (consulta principal)
 * Endpoint público confirmado nos seus JS:
 *   /publico/json/util/localizarPc
 *
 * O parâmetro de CNPJ pode variar.
 * Por padrão tentamos:
 *   entidadeCnpj (1º)
 *   cnpj (2º)
 *   cnpjEntidade (3º)
 *
 * A rota abaixo aceita ?cnpj= com ou sem máscara.
 */
app.get("/api/sigpc/localizar", async (req, res) => {
  try {
    const {
      cnpj,
      tipoConsulta = "2",
      anoConcessao = "2025",
      programaId = "40",
      limit = "40",
      start = "0", // alguns stores usam "start" para paginação
    } = req.query;

    const cnpjDigits = normalizeCnpj(cnpj);

    // Tentativas de nomes de parâmetro para CNPJ (não dá pra garantir sem ver o JSON do FNDE)
    const attempts = [
      { entidadeCnpj: cnpjDigits },
      { cnpj: cnpjDigits },
      { cnpjEntidade: cnpjDigits },
      {}, // sem CNPJ (fallback)
    ];

    let lastErr = null;
    for (const cnpjParam of attempts) {
      try {
        const data = await fndeGet("/publico/json/util/localizarPc", {
          tela: "localizar",
          tipoConsulta,
          anoConcessao,
          programaId,
          limit,
          start,
          ...cnpjParam,
        });

        // Se o usuário passou CNPJ, e a tentativa foi com CNPJ, e retornou algo "utilizável", devolve.
        // Como a estrutura pode variar, a regra é simples:
        // - se veio JSON e não veio erro explícito, devolve.
        if (data && typeof data === "object") {
          return res.json({
            ok: true,
            usedCnpjParam: Object.keys(cnpjParam)[0] || null,
            query: {
              cnpj: cnpjDigits || null,
              tipoConsulta,
              anoConcessao,
              programaId,
              limit,
              start,
            },
            data,
          });
        }
      } catch (e) {
        lastErr = e;
      }
    }

    throw lastErr || new Error("Não consegui consultar localizarPc.");
  } catch (e) {
    res.status(500).json({ ok: false, error: String(e) });
  }
});

/**
 * 2) DETALHAR por pcId + dirigentes (atual e ex)
 * O seu JS usa localizarPc com pcId para popular o painel.
 * Para dirigentes:
 *   /json/dirigente/atualizarDirigenteListarDirigentes?idPc=...&tipoDirigente=A|E
 */
app.get("/api/sigpc/pc/:pcId", async (req, res) => {
  const { pcId } = req.params;

  try {
    // Detalhe: primeiro tenta o endpoint público
    let pcDetail = null;
    let pcDetailSource = "publico";

    try {
      pcDetail = await fndeGet("/publico/json/util/localizarPc", { pcId });
    } catch {
      // fallback caso o ambiente exija outro caminho
      pcDetailSource = "json";
      pcDetail = await fndeGet("/json/util/localizarPc", { pcId });
    }

    // Dirigentes
    let dirigenteAtual = null;
    let exDirigente = null;

    try {
      dirigenteAtual = await fndeGet(
        "/json/dirigente/atualizarDirigenteListarDirigentes",
        { idPc: pcId, tipoDirigente: "A" }
      );
    } catch {}

    try {
      exDirigente = await fndeGet(
        "/json/dirigente/atualizarDirigenteListarDirigentes",
        { idPc: pcId, tipoDirigente: "E" }
      );
    } catch {}

    res.json({
      ok: true,
      pcId,
      pcDetailSource,
      pc: pcDetail,
      dirigenteAtual,
      exDirigente,
    });
  } catch (e) {
    res.status(500).json({ ok: false, error: String(e) });
  }
});

/**
 * 3) HISTÓRICO de análise por concessaoId
 * Endpoint apontado no seu JS:
 *   /json/conferirdocumentos/ConferirDocumentos?concessaoId=...&comando=recuperarHistoricoAnalise
 */
app.get("/api/sigpc/historico/:concessaoId", async (req, res) => {
  const { concessaoId } = req.params;

  try {
    const data = await fndeGet("/json/conferirdocumentos/ConferirDocumentos", {
      concessaoId,
      comando: "recuperarHistoricoAnalise",
    });

    res.json({ ok: true, concessaoId, data });
  } catch (e) {
    res.status(500).json({ ok: false, error: String(e) });
  }
});

/**
 * 4) HEALTHCHECK
 */
app.get("/api/health", (req, res) => {
  res.json({ ok: true, service: "sigpc-scraper", base: BASE });
});

app.listen(PORT, () => {
  console.log(`SIGPC API on port ${PORT}`);
});
