/**

/**
 * Cache simples in-memory (opcional, mas recomendado)
 */
const cache = new Map();
function cacheKey(method, url, paramsOrBody) {
  return `${method}:${url}:${JSON.stringify(paramsOrBody ?? {})}`;
}
async function withCache(key, ttlMs, fn) {
  const now = Date.now();
  const hit = cache.get(key);
  if (hit && (now - hit.time) < ttlMs) return hit.value;

  const value = await fn();
  cache.set(key, { time: now, value });
  return value;
}

/**
 * Helpers
 */
async function fndeGet(path, params = {}, ttlMs = 6 * 60 * 60 * 1000) {
  const url = `${BASE}${path}`;
  const key = cacheKey("GET", url, params);

  return withCache(key, ttlMs, async () => {
    const resp = await http.get(url, { params });
    return resp.data;
  });
}

async function fndePost(path, body = {}, ttlMs = 0) {
  const url = `${BASE}${path}`;
  const key = cacheKey("POST", url, body);

  // por padrão eu deixo POST sem cache (ttl=0),
  // mas você pode passar ttlMs > 0 quando fizer sentido.
  if (ttlMs > 0) {
    return withCache(key, ttlMs, async () => {
      const resp = await http.post(url, body);
      return resp.data;
    });
  }

  const resp = await http.post(url, body);
  return resp.data;
}

/**
 * Normalizações
 */
function cleanCnpj(value) {
  if (!value) return undefined;
  return String(value).replace(/[.\-\/_]/g, "");
}

 *  CONSULTAR SITUAÇÃO (tela ConsultarSituacao)
 * ========================================================= */

/**
 * GET /sigpc/consultar-situacao
 * Query:
 *  - uf + municipio  OU  cnpj
 *  - (opcionais): ano, programa, tipoEntidade, dataEnvio, aptaPagamento
 *
 * Backend:
 *  POST /publico/json/consultasituacao/consultaSituacao
 */
app.get("/sigpc/consultar-situacao", async (req, res) => {
  try {
    const {
      uf, municipio, ano, programa, tipoEntidade, dataEnvio, aptaPagamento, cnpj,
      start, limit
    } = req.query;

    if ((!uf || !municipio) && !cnpj) {
      return res.status(400).json({ error: "Informe uf+municipio ou cnpj." });
    }

    const body = {
      ...(uf ? { uf } : {}),
      ...(municipio ? { municipio } : {}),
      ...(ano ? { ano } : {}),
      ...(programa ? { programa } : {}),
      ...(tipoEntidade ? { tipoEntidade } : {}),
      ...(dataEnvio ? { dataEnvio } : {}),
      ...(aptaPagamento ? { aptaPagamento } : {}),
      ...(cnpj ? { cnpj: cleanCnpj(cnpj) } : {}),
      start: start ? Number(start) : 0,
      limit: limit ? Number(limit) : 25,
      listSize: limit ? Number(limit) : 25, // o JS faz listSize=limit no beforeload
    };

    const data = await fndePost("/publico/json/consultasituacao/consultaSituacao", body, 0);
    res.json(data);
  } catch (e) {
    res.status(502).json({ error: "Falha ao consultar situação", details: String(e?.message ?? e) });
  }
});
